    <table_description>Core network infrastructure table for Mobile Private Networks (MPN) serving enterprise clients.</table_description>
    <table_schema>
    networks (
        network_id VARCHAR(50) PRIMARY KEY,
        network_name VARCHAR(100) NOT NULL,
        enterprise_client VARCHAR(100) NOT NULL,
        location VARCHAR(100),
        created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
    )
    </table_schema>
    <data_dictionary>
    network_id: Unique identifier for each MPN network (e.g., MPN-001)
    network_name: Descriptive name of the network (e.g., Enterprise Network 1)
    enterprise_client: Mexican enterprise client name (e.g., PEMEX, América Móvil, Grupo Bimbo)
    location: Geographic location of the network
    created_at: Network creation timestamp
    </data_dictionary>
    </table>
